! ============================================================
! CONFIGURACIÓN CORE-SW-2 (Switch Core Secundario)
! ============================================================
! Switch: Cisco Catalyst 3650-24PS
! Rol: Core L3 Secundario (Standby)
! Features: EtherChannel, HSRP, Inter-VLAN Routing
! ============================================================

enable
configure terminal
hostname CORE-SW-2

! Desactivar búsqueda DNS
no ip domain-lookup

! ============================================================
! VLANs
! ============================================================
vlan 10
 name ZONA-A
vlan 20
 name ZONA-B
vlan 30
 name ZONA-C
vlan 40
 name ZONA-D
vlan 50
 name ZONA-E
vlan 60
 name ZONA-F
vlan 99
 name MGMT-DMZ
vlan 100
 name NATIVE-MGMT

! ============================================================
! INTERFAZ A ISR-WAN-2 (L3 - Routed Port)
! ============================================================
interface GigabitEthernet1/0/1
 description Enlace a ISR-WAN-2
 no switchport
 ip address 10.0.0.6 255.255.255.252
 no shutdown

! ============================================================
! ETHERCHANNEL A CORE-SW-1 (L3)
! ============================================================
interface Port-channel1
 description EtherChannel L3 a CORE-SW-1
 no switchport
 ip address 10.0.1.2 255.255.255.252
 no shutdown

interface GigabitEthernet1/0/23
 description Miembro EtherChannel
 channel-group 1 mode active
 no shutdown

interface GigabitEthernet1/0/24
 description Miembro EtherChannel
 channel-group 1 mode active
 no shutdown

! ============================================================
! INTERFACES SVI - HSRP + Gateway para cada VLAN
! ============================================================

! VLAN 10 - ZONA A
interface Vlan10
 description Gateway ZONA-A
 ip address 192.168.10.3 255.255.255.128
 standby version 2
 standby 10 ip 192.168.10.1
 standby 10 priority 100
 standby 10 preempt
 standby 10 timers 1 3
 no shutdown

! VLAN 20 - ZONA B
interface Vlan20
 description Gateway ZONA-B
 ip address 192.168.20.3 255.255.255.128
 standby version 2
 standby 20 ip 192.168.20.1
 standby 20 priority 100
 standby 20 preempt
 standby 20 timers 1 3
 no shutdown

! VLAN 30 - ZONA C
interface Vlan30
 description Gateway ZONA-C
 ip address 192.168.30.3 255.255.255.128
 standby version 2
 standby 30 ip 192.168.30.1
 standby 30 priority 100
 standby 30 preempt
 standby 30 timers 1 3
 no shutdown

! VLAN 40 - ZONA D
interface Vlan40
 description Gateway ZONA-D
 ip address 192.168.40.3 255.255.255.128
 standby version 2
 standby 40 ip 192.168.40.1
 standby 40 priority 100
 standby 40 preempt
 standby 40 timers 1 3
 no shutdown

! VLAN 50 - ZONA E
interface Vlan50
 description Gateway ZONA-E
 ip address 192.168.50.3 255.255.255.128
 standby version 2
 standby 50 ip 192.168.50.1
 standby 50 priority 100
 standby 50 preempt
 standby 50 timers 1 3
 no shutdown

! VLAN 60 - ZONA F
interface Vlan60
 description Gateway ZONA-F
 ip address 192.168.60.3 255.255.255.128
 standby version 2
 standby 60 ip 192.168.60.1
 standby 60 priority 100
 standby 60 preempt
 standby 60 timers 1 3
 no shutdown

! VLAN 99 - MGMT/DMZ
interface Vlan99
 description Gateway MGMT
 ip address 10.0.99.3 255.255.255.0
 standby version 2
 standby 99 ip 10.0.99.1
 standby 99 priority 100
 standby 99 preempt
 standby 99 timers 1 3
 no shutdown

! ============================================================
! TRUNKS A SWITCHES DE DISTRIBUCIÓN
! ============================================================
interface GigabitEthernet1/0/2
 description Trunk a DIST-SW-A
 switchport trunk encapsulation dot1q
 switchport trunk allowed vlan 10,20,30,99,100
 switchport trunk native vlan 100
 switchport mode trunk
 switchport nonegotiate
 spanning-tree link-type point-to-point
 no shutdown

interface GigabitEthernet1/0/3
 description Trunk a DIST-SW-B
 switchport trunk encapsulation dot1q
 switchport trunk allowed vlan 40,50,60,99,100
 switchport trunk native vlan 100
 switchport mode trunk
 switchport nonegotiate
 spanning-tree link-type point-to-point
 no shutdown

! ============================================================
! RUTAS ESTÁTICAS
! ============================================================
! Habilitar routing
ip routing

! Ruta por defecto hacia routers WAN
ip route 0.0.0.0 0.0.0.0 10.0.0.5

! Ruta hacia CORE-SW-1 por EtherChannel
ip route 10.0.1.0 255.255.255.252 10.0.1.1

! ============================================================
! CONFIGURACIÓN DHCP RELAY (IP Helper)
! ============================================================
interface Vlan10
 ip helper-address 10.0.99.10

interface Vlan20
 ip helper-address 10.0.99.10

interface Vlan30
 ip helper-address 10.0.99.10

interface Vlan40
 ip helper-address 10.0.99.10

interface Vlan50
 ip helper-address 10.0.99.10

interface Vlan60
 ip helper-address 10.0.99.10

! ============================================================
! SPANNING TREE (Rapid PVST+)
! ============================================================
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60,99 priority 8192

! ============================================================
! CONFIGURACIÓN SSH
! ============================================================
ip domain-name wan.lab
username admin privilege 15 secret admin123
crypto key generate rsa
  1024

line vty 0 15
 login local
 transport input ssh

! ============================================================
! GUARDAR CONFIGURACIÓN
! ============================================================
end
write memory
